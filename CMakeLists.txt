cmake_minimum_required(VERSION 3.16)
project(ar_slam_system VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_TESTS "Build headless tests" ON)
option(BUILD_VIEWERS "Build interactive camera/OpenGL viewers" OFF)
option(BUILD_BENCHMARKS "Build legacy timing executables" OFF)
option(ENABLE_NATIVE_ARCH "Optimise for local CPU" OFF)
option(WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(MSVC)
    add_compile_options(/W4 /utf-8)
    if(WARNINGS_AS_ERRORS)
        add_compile_options(/WX)
    endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra)
    if(WARNINGS_AS_ERRORS)
        add_compile_options(-Werror)
    endif()
    if(ENABLE_NATIVE_ARCH)
        add_compile_options($<$<CONFIG:Release>:-march=native>)
    endif()
endif()
if(ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(cv_components core imgproc features2d video calib3d)
if(BUILD_VIEWERS)
    list(APPEND cv_components highgui videoio)
    find_package(OpenGL REQUIRED)
    find_package(GLEW REQUIRED)
    find_package(glfw3 REQUIRED)
    find_package(glm REQUIRED)
    find_package(Threads REQUIRED)
endif()
if(BUILD_BENCHMARKS AND NOT BUILD_TESTS)
    message(FATAL_ERROR "BUILD_BENCHMARKS requires BUILD_TESTS")
endif()
if(BUILD_BENCHMARKS)
    find_package(Threads REQUIRED)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS ${cv_components})
message(STATUS "OpenCV version: ${OpenCV_VERSION}")
add_subdirectory(src)
if(BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
